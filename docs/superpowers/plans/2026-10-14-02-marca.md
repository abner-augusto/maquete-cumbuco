## Global Constraints

- O nome "Alternativa" é confidencial até o compliance: a prancha fica em `cache/` (gitignored), leva `noindex` e não é publicada.
- O repo não copia arquivo do cliente: os logos são lidos de `TERRENO_DIR`; o que entra no git são números derivados (`paleta.json`) e o CSS gerado.
- Nenhuma cor do tema é digitada à mão: `tokens.css` sai de `paleta.json` por `scripts/marca/tokens.py`.
- O tema tem exatamente as chaves `--tema-*` do `temas/manoa.css` do espelho, nos dois modos (`[data-tema="alternativa"]` e `[data-tema="alternativa"][data-cor="escuro"]`). Cores de status ficam fora da marca (vivem no `base.css` do front).
- Contraste WCAG: texto ≥ 4,5:1 sobre o papel do modo; `tinta` ≥ 7:1.
- Fontes abertas (OFL) do Google Fonts. Candidatas: serifada Cinzel ou Marcellus; sans Jost ou Montserrat. Padrão até o gate: Cinzel + Jost.
- Docs, UI e commits em português do Brasil.

## Review Focus

1. **Ouro sobre creme não passa em contraste de texto.** O ouro medido é claro demais para texto; o gerador precisa escurecer até 4,5:1, não usar o ouro cru. Teste na Task 3.

3. **Chave faltando no modo escuro** faz o front herdar a cor do claro sem aviso. Teste na Task 3 (conjuntos de chaves iguais e iguais aos do Manoa).
4. **Nomes de arquivo com espaço, parênteses e acento** (`Estudo Logo Joao Carlos/WhatsApp Image ... (1).jpeg`) precisam abrir. Teste na Task 4.
5. **A prancha não pode vazar:** destino dentro de `cache/` e `noindex`. Teste na Task 4.

```python
"""Aritmética de cor em sRGB 8 bits e contraste WCAG 2.x."""

Cor = tuple[int, int, int]


def hex_para_rgb(h: str) -> Cor:
    h = h.lstrip("#")
    return (int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16))


def rgb_para_hex(c: Cor) -> str:
    return "#{:02X}{:02X}{:02X}".format(*c)


def misturar(a: Cor, b: Cor, t: float) -> Cor:
    return tuple(round(x + (y - x) * t) for x, y in zip(a, b))  # type: ignore[return-value]


def _linear(canal: int) -> float:
    c = canal / 255
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def luminancia(c: Cor) -> float:
    r, g, b = (_linear(x) for x in c)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def contraste(a: Cor, b: Cor) -> float:
    la, lb = sorted((luminancia(a), luminancia(b)), reverse=True)
    return (la + 0.05) / (lb + 0.05)


def ajustar_contraste(cor: Cor, fundo: Cor, minimo: float) -> Cor:
    """Escurece (fundo claro) ou clareia (fundo escuro) a cor, mantendo o matiz, até o contraste mínimo."""
    alvo: Cor = (0, 0, 0) if luminancia(fundo) > 0.18 else (255, 255, 255)
    for passo in range(101):
        nova = misturar(cor, alvo, passo / 100)
        if contraste(nova, fundo) >= minimo:
            return nova
    raise ValueError(f"contraste {minimo} impossível sobre {rgb_para_hex(fundo)}")
```

**Estratégia:** dourado = matiz 20°–55°, saturação ≥ 0,25, valor ≥ 0,25. Os pixels dourados de todas as imagens são ordenados por luminância; a mediana por canal de três faixas dá o bronze (5–20%), o ouro (40–60%) e o ouro claro (80–95%). As faixas evitam os extremos, onde moram o brilho e o artefato do JPEG. O creme é a mediana dos pixels com saturação < 0,15 e valor > 0,85 das imagens claras; o preto, a mediana dos pixels com valor < 0,12 das imagens escuras. Imagem clara ou escura se decide pela mediana do valor (> 0,5 é clara).

```python
"""Paleta medida nos logos raster. Uso: uv run python -m scripts.marca.paleta"""

import json
import sys
from dataclasses import asdict, dataclass
from pathlib import Path

import numpy as np
from PIL import Image

from scripts.comum.ambiente import RAIZ, terreno_dir

from .cores import rgb_para_hex

_MIDIA = "6-PROJETOS/09-MIDIA"
_ESTUDOS = f"{_MIDIA}/Estudo Logo Joao Carlos"
FONTES_LOGO = {
    "P0": f"{_MIDIA}/LOGO Preliminar.jpeg",
    "E1": f"{_ESTUDOS}/WhatsApp Image 2026-10-14 at 11.05.31 (1).jpeg",
    "E2": f"{_ESTUDOS}/WhatsApp Image 2026-10-14 at 11.05.31.jpeg",
    "E3": f"{_ESTUDOS}/WhatsApp Image 2026-10-14 at 11.05.32 (1).jpeg",
    "E4": f"{_ESTUDOS}/WhatsApp Image 2026-10-14 at 11.05.32 (2).jpeg",
    "E5": f"{_ESTUDOS}/WhatsApp Image 2026-10-14 at 11.05.32 (3).jpeg",
    "E6": f"{_ESTUDOS}/WhatsApp Image 2026-10-14 at 11.05.32.jpeg",
}
DESTINO = RAIZ / "marca" / "paleta.json"


@dataclass(frozen=True)
class Paleta:
    ouro_claro: str
    ouro: str
    bronze: str
    creme: str
    preto: str

    def para_json(self) -> dict:
        return asdict(self)

    @classmethod
    def carregar(cls, caminho: Path) -> "Paleta":
        return cls(**json.loads(caminho.read_text(encoding="utf-8")))


def pixels(img: Image.Image, lado: int = 256) -> np.ndarray:
    img = img.convert("RGB")
    img.thumbnail((lado, lado))
    return np.asarray(img, dtype=np.float64).reshape(-1, 3)


def _hsv(px: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    r, g, b = (px[:, i] / 255 for i in range(3))
    mx, mn = np.max(px / 255, axis=1), np.min(px / 255, axis=1)
    delta = mx - mn
    h = np.zeros_like(mx)
    m = delta > 0
    rm, gm, bm = m & (mx == r), m & (mx == g) & (mx != r), m & (mx == b) & (mx != r) & (mx != g)
    h[rm] = (60 * ((g[rm] - b[rm]) / delta[rm])) % 360
    h[gm] = 60 * ((b[gm] - r[gm]) / delta[gm]) + 120
    h[bm] = 60 * ((r[bm] - g[bm]) / delta[bm]) + 240
    s = np.where(mx > 0, delta / np.where(mx > 0, mx, 1), 0)
    return h, s, mx


def _mediana(px: np.ndarray, nome: str) -> str:
    if len(px) == 0:
        raise ValueError(f"nenhum pixel de {nome} nas imagens")
    return rgb_para_hex(tuple(int(round(v)) for v in np.median(px, axis=0)))


def medir(imagens: list[Image.Image]) -> Paleta:
    todas = [pixels(i) for i in imagens]
    claras = [p for p in todas if np.median(_hsv(p)[2]) > 0.5]
    escuras = [p for p in todas if np.median(_hsv(p)[2]) <= 0.5]

    px = np.concatenate(todas)
    h, s, v = _hsv(px)
    dourado = px[(h >= 20) & (h <= 55) & (s >= 0.25) & (v >= 0.25)]
    if len(dourado) == 0:
        raise ValueError("nenhum pixel de ouro nas imagens")
    ordem = dourado[np.argsort(dourado @ np.array([0.2126, 0.7152, 0.0722]))]
    n = len(ordem)
    faixa = lambda a, b: ordem[int(n * a) : max(int(n * b), int(n * a) + 1)]  # noqa: E731

    creme = np.concatenate(claras) if claras else np.empty((0, 3))
    if len(creme):
        _, sc, vc = _hsv(creme)
        creme = creme[(sc < 0.15) & (vc > 0.85)]
    preto = np.concatenate(escuras) if escuras else np.empty((0, 3))
    if len(preto):
        preto = preto[_hsv(preto)[2] < 0.12]

    return Paleta(
        ouro_claro=_mediana(faixa(0.80, 0.95), "ouro claro"),
        ouro=_mediana(faixa(0.40, 0.60), "ouro"),
        bronze=_mediana(faixa(0.05, 0.20), "bronze"),
        creme=_mediana(creme, "creme"),
        preto=_mediana(preto, "preto"),
    )


def main() -> int:
    terreno = terreno_dir()
    imagens = [Image.open(terreno / rel) for rel in FONTES_LOGO.values()]
    p = medir(imagens)
    DESTINO.parent.mkdir(parents=True, exist_ok=True)
    DESTINO.write_text(json.dumps(p.para_json(), indent=1) + "\n", encoding="utf-8")
    print(p)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Tema do front (formato do temas/manoa.css do espelho) derivado da paleta. Uso: uv run python -m scripts.marca.tokens"""

import sys

from scripts.comum.ambiente import RAIZ

from .cores import Cor, ajustar_contraste, contraste, hex_para_rgb, misturar, rgb_para_hex
from .paleta import DESTINO as PALETA_JSON
from .paleta import Paleta

CHAVES = (
    "papel", "papel-2", "areia", "filete", "filete-forte", "tinta", "tinta-2", "tinta-3",
    "acento", "acento-contraste", "acento-hover", "caramelo", "sol", "sol-texto", "sol-fundo",
    "sol-sobre-tinta", "ceu", "campo",
)
FONTES_PADRAO = {"display": "Cinzel", "texto": "Jost", "rotulo": "Jost"}
DESTINO = RAIZ / "marca" / "tokens.css"
_BRANCO: Cor = (255, 255, 255)
_PRETO: Cor = (0, 0, 0)


def _texto_sobre(fundo: Cor, *candidatas: Cor) -> Cor:
    return next((c for c in candidatas if contraste(c, fundo) >= 4.5), ajustar_contraste(candidatas[0], fundo, 4.5))


def _claro(c: Cor, k: Cor, o: Cor, oc: Cor, b: Cor) -> dict[str, Cor]:
    papel = c
    acento = ajustar_contraste(b, papel, 4.5)
    return {
        "papel": papel,
        "papel-2": misturar(c, b, 0.06),
        "areia": misturar(c, o, 0.22),
        "filete": misturar(c, b, 0.18),
        "filete-forte": misturar(c, b, 0.35),
        "tinta": ajustar_contraste(misturar(k, b, 0.12), papel, 7),
        "tinta-2": ajustar_contraste(misturar(k, b, 0.45), papel, 4.5),
        "tinta-3": ajustar_contraste(misturar(k, b, 0.60), papel, 4.5),
        "acento": acento,
        "acento-contraste": _texto_sobre(acento, c, _BRANCO),
        "acento-hover": misturar(acento, k, 0.15),
        "caramelo": o,
        "sol": oc,
        "sol-texto": ajustar_contraste(o, papel, 4.5),
        "sol-fundo": misturar(c, oc, 0.25),
        "sol-sobre-tinta": oc,
        "ceu": misturar(c, b, 0.10),
        "campo": _BRANCO,
    }


def _escuro(c: Cor, k: Cor, o: Cor, oc: Cor, b: Cor) -> dict[str, Cor]:
    papel = misturar(k, b, 0.06)
    acento = ajustar_contraste(oc, papel, 4.5)
    return {
        "papel": papel,
        "papel-2": misturar(k, b, 0.12),
        "areia": misturar(k, o, 0.18),
        "filete": misturar(k, b, 0.22),
        "filete-forte": misturar(k, b, 0.40),
        "tinta": ajustar_contraste(c, papel, 7),
        "tinta-2": ajustar_contraste(misturar(c, b, 0.30), papel, 4.5),
        "tinta-3": ajustar_contraste(misturar(c, b, 0.45), papel, 4.5),
        "acento": acento,
        "acento-contraste": _texto_sobre(acento, k, _PRETO),
        "acento-hover": misturar(acento, c, 0.20),
        "caramelo": o,
        "sol": oc,
        "sol-texto": ajustar_contraste(oc, papel, 4.5),
        "sol-fundo": misturar(k, o, 0.25),
        "sol-sobre-tinta": ajustar_contraste(b, c, 3),
        "ceu": misturar(k, b, 0.10),
        "campo": misturar(k, _PRETO, 0.30),
    }


def tema(p: Paleta) -> dict[str, dict[str, str]]:
    cores = [hex_para_rgb(x) for x in (p.creme, p.preto, p.ouro, p.ouro_claro, p.bronze)]
    return {
        modo: {k: rgb_para_hex(v) for k, v in f(*cores).items()}
        for modo, f in (("claro", _claro), ("escuro", _escuro))
    }


def _sombras(rgb: str, a: tuple[float, float, float, float, float, float]) -> list[str]:
    return [
        f"  --tema-sombra-1: 0 1px 2px rgba({rgb}, {a[0]}), 0 4px 12px rgba({rgb}, {a[1]});",
        f"  --tema-sombra-2: 0 -1px 0 rgba({rgb}, {a[2]}), 0 -8px 28px rgba({rgb}, {a[3]});",
        f"  --tema-sombra-3: 0 2px 4px rgba({rgb}, {a[4]}), 0 10px 22px rgba({rgb}, {a[5]});",
    ]


def css(t: dict[str, dict[str, str]], fontes: dict[str, str], nome: str = "alternativa") -> str:
    tinta = ", ".join(str(x) for x in hex_para_rgb(t["claro"]["tinta"]))
    linhas = [
        f"/* temas/{nome} — gerado por scripts/marca/tokens.py a partir de marca/paleta.json. Não editar à mão. */",
        f'[data-tema="{nome}"] {{',
        *[f"  --tema-{k}: {t['claro'][k]};" for k in CHAVES],
        "",
        f'  --tema-fonte-display: "{fontes["display"]}", Georgia, serif;',
        f'  --tema-fonte-texto: "{fontes["texto"]}", system-ui, sans-serif;',
        f'  --tema-fonte-rotulo: "{fontes["rotulo"]}", system-ui, sans-serif;',
        "",
        *_sombras(tinta, (0.10, 0.08, 0.06, 0.14, 0.10, 0.14)),
        "}",
        "",
        f'[data-tema="{nome}"][data-cor="escuro"] {{',
        *[f"  --tema-{k}: {t['escuro'][k]};" for k in CHAVES],
        "",
        *_sombras("0, 0, 0", (0.35, 0.28, 0.3, 0.45, 0.35, 0.45)),
        "}",
    ]
    return "\n".join(linhas) + "\n"


def main() -> int:
    t = tema(Paleta.carregar(PALETA_JSON))
    DESTINO.write_text(css(t, FONTES_PADRAO), encoding="utf-8")
    print(f"{DESTINO} ({len(CHAVES)} cores × 2 modos)")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def miniatura(caminho: Path, lado: int = 900) -> str:
    img = Image.open(caminho).convert("RGB")

    buf = io.BytesIO()
    img.save(buf, format="JPEG", quality=82)
```
