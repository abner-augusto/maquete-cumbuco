**Goal:** Pôr clube, portaria e rotatória (modelos do SketchUp já convertidos) no lugar certo da cena, sem brigar com a placa, e gerar `dados/marcos.json` com posição, galeria e coordenada de cada marco, incluindo o estande como ponto.

```python
def umeyama(a: np.ndarray, b: np.ndarray, escala: bool = True) -> Semelhanca:
    """Semelhança de mínimos quadrados a → b, sem espelho; com `escala=False`, rígida (s = 1)."""
    ma, mb = a.mean(0), b.mean(0)
    da, db = a - ma, b - mb
    u, d, vt = np.linalg.svd(db.T @ da / len(a))
    sinal = np.diag([1.0, np.sign(np.linalg.det(u @ vt))])
    r = u @ sinal @ vt
    s = float(np.trace(np.diag(d) @ sinal) / (da ** 2).sum(1).mean()) if escala else 1.0
    t = mb - s * r @ ma
    return Semelhanca(s, math.atan2(r[1, 0], r[0, 0]), float(t[0]), float(t[1]))
```

```python
"""Encaixe dos marcos na planta de venda: dois pares de pontos (ilha↔círculo, clube↔quadra) dão a
transformação rígida; ICP com o chão da portaria a refina. Python puro: o Blender só extrai e aplica.

Uso: uv run python -m scripts.marcos.encaixe
"""

import math
import tomllib
from dataclasses import dataclass, field

import numpy as np
from scipy.spatial import cKDTree

from scripts.comum.ambiente import RAIZ
from scripts.entorno.masterplan import Semelhanca, umeyama

CFG_TOML = RAIZ / "scripts" / "marcos" / "marcos.toml"


@dataclass(frozen=True)
class MarcosCfg:
    pasta: str
    clube: str
    portaria: str
    pivo_clube: list[float]
    giro_clube: float
    edificacao: list[str]
    margem_portaria: float
    raio_icp: float
    corte_icp: float
    residuo_max: float
    tol_raio: float
    tol_dist: float
    cobertura_min: float
    kmz_estande: str

    @classmethod
    def carregar(cls) -> "MarcosCfg":
        return cls(**tomllib.loads(CFG_TOML.read_text(encoding="utf-8")))


def girar(pts, pivo: tuple[float, float], graus: float) -> np.ndarray:
    a = math.radians(graus)
    r = np.array([[math.cos(a), -math.sin(a)], [math.sin(a), math.cos(a)]])
    p = np.asarray(pts, dtype=float) - pivo
    return p @ r.T + pivo


def escolher_circulo(circulos: list[tuple[float, float, float]], raio: float, dist_clube: float,
                     quadra: tuple[float, float], tol_raio: float = 3.0, tol_dist: float = 10.0
                     ) -> tuple[float, float, float]:
    """O círculo do DWG com o raio da ilha do GLB e à distância ilha–clube da quadra central."""
    bons = [c for c in circulos if abs(c[2] - raio) <= tol_raio
            and abs(math.hypot(c[0] - quadra[0], c[1] - quadra[1]) - dist_clube) <= tol_dist]
    if len(bons) != 1:
        raise SystemExit(f"círculo da rotatória: {len(bons)} candidatos (raio {raio:.1f} m, "
                         f"distância ao clube {dist_clube:.1f} m) entre {circulos}")
    return bons[0]


def ancora(ilha, clube, circulo, quadra) -> Semelhanca:
    """Transformação rígida que leva ilha→círculo e o rumo ilha→clube ao rumo círculo→quadra."""
    ilha, clube, circulo, quadra = (np.asarray(p, dtype=float) for p in (ilha, clube, circulo, quadra))
    de, para = clube - ilha, quadra - circulo
    theta = math.atan2(para[1], para[0]) - math.atan2(de[1], de[0])
    c, s = math.cos(theta), math.sin(theta)
    t = circulo - np.array([[c, -s], [s, c]]) @ ilha
    return Semelhanca(1.0, theta, float(t[0]), float(t[1]))


def icp_rigido(origem: np.ndarray, destino: np.ndarray, inicial: Semelhanca, iteracoes: int = 30,
               corte: float = 5.0) -> tuple[Semelhanca, float]:
    """ICP rígido a partir de `inicial`; par mais longe que `corte` não entra (sobra de um dos lados)."""
    arvore = cKDTree(destino)
    t = inicial
    for _ in range(iteracoes):
        d, i = arvore.query(t.aplicar(origem))
        ok = d <= corte
        if ok.sum() < 3:
            break
        t = umeyama(origem[ok], destino[i[ok]], escala=False)
    d, _ = arvore.query(t.aplicar(origem))
    return t, float(np.median(d[d <= corte])) if (d <= corte).any() else math.inf
```

```python
"""Extrai dos GLBs fonte dos marcos só o que o encaixe precisa. Roda dentro do Blender:

    blender -b --factory-startup --python-exit-code 1 -P extrair_marcos.py -- destino.json PREFIXO[,PREFIXO] a.glb [b.glb ...]

Caixa de cada objeto e triângulos de chão (normal para cima, perto de z = 0), em coordenadas do Blender.
"""

import json
import sys
from pathlib import Path

import bpy

args = sys.argv[sys.argv.index("--") + 1:]
DESTINO, PREFIXOS, GLBS = Path(args[0]), args[1].split(","), [Path(a) for a in args[2:]]

saida = {}
for glb in GLBS:
    bpy.ops.wm.read_factory_settings(use_empty=True)
    bpy.ops.import_scene.gltf(filepath=str(glb))
    objetos, chao = {}, {}
    for ob in bpy.context.scene.objects:
        if ob.type != "MESH":
            continue
        mw = ob.matrix_world
        vs = [mw @ v.co for v in ob.data.vertices]
        if not vs:
            continue
        objetos[ob.name] = [round(min(v.x for v in vs), 3), round(min(v.y for v in vs), 3), round(min(v.z for v in vs), 3),
                            round(max(v.x for v in vs), 3), round(max(v.y for v in vs), 3), round(max(v.z for v in vs), 3)]
        nrm = mw.to_3x3()
        tris = []
        for p in ob.data.polygons:
            n = (nrm @ p.normal).normalized()
            pts = [vs[i] for i in p.vertices]
            zm = sum(q.z for q in pts) / len(pts)
            if n.z > 0.9 and -0.3 <= zm <= 0.6:
                for k in range(1, len(pts) - 1):  # leque: o polígono vira triângulos
                    a, b, c = pts[0], pts[k], pts[k + 1]
                    tris.append([round(a.x, 3), round(a.y, 3), round(b.x, 3), round(b.y, 3), round(c.x, 3), round(c.y, 3)])
        if tris:
            chao[ob.name] = tris
    saida[glb.name] = {"objetos": objetos, "chao": chao}

faltam = [p for p in PREFIXOS if not any(n.startswith(p) for d in saida.values() for n in d["objetos"])]
if faltam:
    print("RESUMO " + json.dumps({"erro": f"nenhum objeto com prefixo {', '.join(faltam)} em {[g.name for g in GLBS]}"}))
    sys.exit(0)
DESTINO.parent.mkdir(parents=True, exist_ok=True)
DESTINO.write_text(json.dumps(saida), encoding="utf-8")
print("RESUMO " + json.dumps({k: {"objetos": len(v["objetos"]), "chao": sum(len(t) for t in v["chao"].values())}
                               for k, v in saida.items()}))
```

```python
"""Driver dos scripts de Blender dos marcos. Uso: uv run python -m scripts.marcos.blender (Task 5)."""

import json
from pathlib import Path

from scripts.comum.ambiente import RAIZ
from scripts.loteamento.blender import rodar_blender

EXTRAIR = RAIZ / "scripts" / "blender" / "extrair_marcos.py"
FONTE_JSON = RAIZ / "cache" / "cena" / "marcos-fonte.json"


def rodar_extracao(blender: Path, glbs: list[Path], destino: Path, prefixos: list[str]) -> dict:
    resumo = rodar_blender(blender, EXTRAIR, str(destino), ",".join(prefixos), *[str(g) for g in glbs])
    if "erro" in resumo:
        raise RuntimeError(resumo["erro"])
    return json.loads(destino.read_text(encoding="utf-8"))


def extrair() -> int:
    """`uv run python -m scripts.marcos.blender --extrair`: caixas e chão dos GLBs fonte."""
    from scripts.comum.ambiente import ler_env, terreno_dir
    from scripts.loteamento.blender import BLENDER_PADRAO

    from .encaixe import MarcosCfg

    c = MarcosCfg.carregar()
    pasta = terreno_dir() / c.pasta
    blender = Path(ler_env(RAIZ / ".env").get("BLENDER", BLENDER_PADRAO))
    d = rodar_extracao(blender, [pasta / c.clube, pasta / c.portaria], FONTE_JSON, ["CLUBE_", "PORTARIA_", "ROTATORIA_"])
    print({k: (len(v["objetos"]), sum(len(t) for t in v["chao"].values())) for k, v in d.items()})
    return 0


if __name__ == "__main__":
    import sys

    sys.exit(extrair())
```

```python
def quadra_do_clube(chao_plato, plato) -> Polygon:
    """A maior parte do ChaoPlato que não toca a borda do platô: a quadra sem lote do meio."""
    borda = plato.exterior if isinstance(plato, Polygon) else unary_union([p.exterior for p in partes(plato)])
    interiores = [p for p in partes(chao_plato) if not p.buffer(0.01).intersects(borda)]
    if not interiores:
        raise SystemExit(f"quadra do clube: nenhuma parte interior do ChaoPlato (partes: "
                         f"{[round(p.area) for p in partes(chao_plato)][:10]} m²)")
    return max(interiores, key=lambda p: p.area)


def pegada_de_triangulos(tris) -> object:
    return unary_union([Polygon([(t[0], t[1]), (t[2], t[3]), (t[4], t[5])]).buffer(0.001) for t in tris])


def pegada_portaria(caixas: list[list[float]], margem: float) -> Polygon:
    pts = [(x, y) for x0, y0, _, x1, y1, _ in caixas for x, y in ((x0, y0), (x1, y0), (x1, y1), (x0, y1))]
    return MultiPoint(pts).convex_hull.buffer(margem, join_style="mitre")


def cobertura(chao, quadra) -> float:
    return chao.intersection(quadra).area / quadra.area


def _centro(caixas) -> np.ndarray:
    x0 = min(c[0] for c in caixas); y0 = min(c[1] for c in caixas)
    x1 = max(c[3] for c in caixas); y1 = max(c[4] for c in caixas)
    return np.array([(x0 + x1) / 2, (y0 + y1) / 2])


def _borda(g, passo: float = 1.0) -> np.ndarray:
    from scripts.entorno.masterplan import amostrar

    return np.vstack([amostrar(np.array(anel.coords[:-1]), passo) for p in partes(g)
                      for anel in [p.exterior, *p.interiors] if len(anel.coords) > 3])


def _chao_clube(tris, pivo, giro, t: Semelhanca) -> object:
    """Triângulos de chão do clube no referencial local: desfaz o giro e aplica a transformação."""
    pts = np.asarray(tris, dtype=float).reshape(-1, 2)
    pts = t.aplicar(girar(pts, pivo, -giro))
    return pegada_de_triangulos(pts.reshape(-1, 6).tolist())


def encaixar(fonte: dict, cfg: MarcosCfg, circulos, pecas: dict, plato, cota: float, alturas: dict,
             vias_dwg) -> dict:
    clube_d, port_d = fonte[cfg.clube], fonte[cfg.portaria]
    cx = [c for n, c in clube_d["objetos"].items() if n.startswith("CLUBE_")]
    ilha_cx = [c for n, c in port_d["objetos"].items() if n.startswith("ROTATORIA_")]
    port_cx = [c for n, c in port_d["objetos"].items()
               if n.startswith("PORTARIA_") and n.split("_", 1)[1].split(".")[0] in cfg.edificacao]
    pivo = tuple(cfg.pivo_clube)
    clube_c = girar(_centro(cx)[None], pivo, -cfg.giro_clube)[0]
    ilha_c = _centro(ilha_cx)
    raio_ilha = max(max(c[3] for c in ilha_cx) - min(c[0] for c in ilha_cx),
                    max(c[4] for c in ilha_cx) - min(c[1] for c in ilha_cx)) / 2
    quadra = quadra_do_clube(pecas["ChaoPlato"], plato)
    qc = (quadra.centroid.x, quadra.centroid.y)
    circ = escolher_circulo(circulos, raio_ilha, float(np.hypot(*(ilha_c - clube_c))), qc, cfg.tol_raio, cfg.tol_dist)
    inicial = ancora(ilha_c, clube_c, circ[:2], qc)
    chao_port = pegada_de_triangulos([t for n, ts in port_d["chao"].items() if n.startswith("PORTARIA_PISO") for t in ts])
    destino = _borda(vias_dwg.intersection(Point(circ[:2]).buffer(cfg.raio_icp)))
    t, residuo = icp_rigido(_borda(chao_port), destino, inicial, corte=cfg.corte_icp)
    print(f"encaixe: âncora {math.degrees(inicial.theta):+.2f}° → ICP {math.degrees(t.theta):+.2f}°, "
          f"({t.tx:.2f}; {t.ty:.2f}) m, resíduo mediano {residuo:.2f} m, círculo r={circ[2]} em {circ[:2]}")
    if residuo > cfg.residuo_max:
        raise SystemExit(f"encaixe: resíduo {residuo:.2f} m acima de {cfg.residuo_max} m")
    clube_final = t.aplicar(clube_c[None])[0]
    if not quadra.contains(Point(clube_final)):
        raise SystemExit(f"encaixe: o clube caiu fora da quadra central ({clube_final.round(1)})")
    cob = cobertura(_chao_clube([tr for ts in clube_d["chao"].values() for tr in ts], pivo, cfg.giro_clube, t), quadra)
    print(f"encaixe: piso do clube cobre {cob:.0%} da quadra ({quadra.area:.0f} m²)")
    if cob < cfg.cobertura_min:
        raise SystemExit(f"encaixe: piso do clube cobre {cob:.0%} da quadra (mínimo {cfg.cobertura_min:.0%})")
    cantos = t.aplicar([[x, y] for c in port_cx for x, y in ((c[0], c[1]), (c[3], c[1]), (c[3], c[4]), (c[0], c[4]))])
    port_peg = MultiPoint([tuple(p) for p in cantos]).convex_hull.buffer(cfg.margem_portaria, join_style="mitre")
    ilha_peg = Point(circ[:2]).buffer(circ[2], 128)
    tj = {"s": 1.0, "theta": t.theta, "tx": t.tx, "ty": t.ty}
    anel = lambda g: [[round(x, 3), round(y, 3)] for x, y in g.exterior.coords[:-1]]  # noqa: E731
    return {"cota": cota, "marcos": {
        "Clube": {"fonte": cfg.clube, "prefixo": "CLUBE_", "giro": [*pivo, -cfg.giro_clube], "transf": tj,
                  "z": round(cota + alturas["ChaoPlato"], 3), "pegada": anel(quadra)},
        "Portaria": {"fonte": cfg.portaria, "prefixo": "PORTARIA_", "giro": None, "transf": tj,
                     "z": round(cota, 3), "pegada": anel(port_peg)},
        "Rotatoria": {"fonte": cfg.portaria, "prefixo": "ROTATORIA_", "giro": None, "transf": tj,
                      "z": round(cota, 3), "pegada": anel(ilha_peg)},
    }}
```

```python
"""Põe os marcos no lugar e exporta. Roda dentro do Blender:

    blender -b --factory-startup --python-exit-code 1 -P gerar_marcos.py -- marcos.json saida.glb [previa.png]

Para cada marco: importa a fonte, fica com os objetos do prefixo, desfaz o giro (clube), aplica a
transformação rígida e a cota, apaga as faces cujo centro está fora da pegada e junta tudo num objeto.
"""

import json
import math
import sys
from pathlib import Path

import bmesh
import bpy
import numpy as np

args = sys.argv[sys.argv.index("--") + 1:]
MARCOS, GLB = Path(args[0]), Path(args[1])
PREVIA = Path(args[2]) if len(args) > 2 else None


def dentro(px: np.ndarray, py: np.ndarray, anel: list) -> np.ndarray:
    """Par/ímpar vetorizado: pontos (px, py) dentro do anel."""
    r = np.zeros(px.shape, dtype=bool)
    xs, ys = np.array([p[0] for p in anel]), np.array([p[1] for p in anel])
    for i in range(len(anel)):
        x0, y0, x1, y1 = xs[i - 1], ys[i - 1], xs[i], ys[i]
        cruza = (y0 > py) != (y1 > py)
        with np.errstate(divide="ignore", invalid="ignore"):
            xi = (x1 - x0) * (py - y0) / (y1 - y0) + x0
        r ^= cruza & (px < xi)
    return r


def girar(x, y, cx, cy, graus):
    a = math.radians(graus)
    dx, dy = x - cx, y - cy
    return cx + dx * math.cos(a) - dy * math.sin(a), cy + dx * math.sin(a) + dy * math.cos(a)


def marco(nome: str, m: dict) -> bpy.types.Object:
    antes = set(bpy.data.objects)
    bpy.ops.import_scene.gltf(filepath=m["fonte"])
    novos = [o for o in bpy.data.objects if o not in antes]
    for o in novos:
        if o.type != "MESH" or not o.name.startswith(m["prefixo"]):
            bpy.data.objects.remove(o)
    obs = [o for o in bpy.data.objects if o not in antes]
    t = m["transf"]
    c, s = math.cos(t["theta"]), math.sin(t["theta"])
    for o in obs:
        o.data.transform(o.matrix_world)  # aplica a matriz do import (y-up → z-up) na malha
        o.matrix_world.identity()
        co = np.empty(len(o.data.vertices) * 3)
        o.data.vertices.foreach_get("co", co)
        co = co.reshape(-1, 3)
        x, y = co[:, 0], co[:, 1]
        if m["giro"]:
            x, y = girar(x, y, *m["giro"])
        co[:, 0], co[:, 1] = c * x - s * y + t["tx"], s * x + c * y + t["ty"]
        co[:, 2] += m["z"]
        o.data.vertices.foreach_set("co", co.ravel())
        o.data.update()
        bm = bmesh.new()
        bm.from_mesh(o.data)
        cen = np.array([f.calc_center_median()[:2] for f in bm.faces]) if bm.faces else np.zeros((0, 2))
        fora = ~dentro(cen[:, 0], cen[:, 1], m["pegada"]) if len(cen) else np.zeros(0, dtype=bool)
        bmesh.ops.delete(bm, geom=[f for f, k in zip(bm.faces, fora) if k], context="FACES")
        bmesh.ops.delete(bm, geom=[v for v in bm.verts if not v.link_faces], context="VERTS")
        bm.to_mesh(o.data)
        bm.free()
    obs = [o for o in obs if len(o.data.polygons)]
    bpy.ops.object.select_all(action="DESELECT")
    for o in obs:
        o.select_set(True)
    bpy.context.view_layer.objects.active = obs[0]
    if len(obs) > 1:
        bpy.ops.object.join()
    ob = bpy.context.view_layer.objects.active
    ob.name = ob.data.name = nome
    return ob


def main() -> None:
    bpy.ops.wm.read_factory_settings(use_empty=True)
    dados = json.loads(MARCOS.read_text(encoding="utf-8"))
    obs = [marco(n, m) for n, m in dados["marcos"].items()]
    bpy.ops.object.select_all(action="DESELECT")
    for o in obs:
        o.select_set(True)
    GLB.parent.mkdir(parents=True, exist_ok=True)
    bpy.ops.export_scene.gltf(filepath=str(GLB), export_format="GLB", use_selection=True, export_apply=True,
                              export_yup=True)
    if PREVIA:
        cena = bpy.context.scene
        try:
            cena.render.engine = "BLENDER_WORKBENCH"
        except TypeError:
            pass
        xs = [v.co.x for o in obs for v in o.data.vertices]
        ys = [v.co.y for o in obs for v in o.data.vertices]
        cam = bpy.data.objects.new("Camera", bpy.data.cameras.new("Camera"))
        cena.collection.objects.link(cam)
        cam.data.type = "ORTHO"
        cam.data.ortho_scale = max(max(xs) - min(xs), max(ys) - min(ys)) * 1.1
        cam.location = ((min(xs) + max(xs)) / 2, (min(ys) + max(ys)) / 2, 500.0)
        cam.data.clip_end = 2000.0
        cena.camera = cam
        cena.render.resolution_x = cena.render.resolution_y = 1400
        cena.render.filepath = str(PREVIA)
        bpy.ops.render.render(write_still=True)
    resumo = {}
    for o in obs:
        vs = [v.co for v in o.data.vertices]
        resumo[o.name] = {"x": [round(min(v.x for v in vs), 3), round(max(v.x for v in vs), 3)],
                          "z": [round(min(v.z for v in vs), 3), round(max(v.z for v in vs), 3)],
                          "triangulos": sum(len(p.vertices) - 2 for p in o.data.polygons)}
    resumo["glb_mb"] = round(GLB.stat().st_size / 1e6, 2)
    print("RESUMO " + json.dumps(resumo))


main()
```

```python
"""dados/marcos.json (contrato com o front) e fotos da galeria reduzidas.

Uso: uv run python -m scripts.marcos.dados
"""

import json
import re
import shutil
import sys
import zipfile
from pathlib import Path

from PIL import Image
from shapely.geometry import Polygon

from scripts.comum.ambiente import RAIZ, terreno_dir
from scripts.entorno.osm import Projecao

from .encaixe import DESTINO as MARCOS_JSON
from .encaixe import MarcosCfg

CURADORIA = RAIZ / "scripts" / "marcos" / "curadoria.json"
DESTINO = RAIZ / "dados" / "marcos.json"
FOTOS = RAIZ / "dados" / "marcos"
PLANTA_KMZ = "files/planta_humanizada.jpg"


def estande_do_kmz(kml: str) -> list[tuple[float, float]]:
    m = re.search(r"<name>Estande \(edificacao\)</name>.*?<coordinates>(.*?)</coordinates>", kml, re.S)
    if not m:
        raise SystemExit("KMZ do estande sem o polígono 'Estande (edificacao)'")
    pts = [tuple(float(v) for v in c.split(",")[:2]) for c in m.group(1).split()]
    return pts[:-1] if pts[0] == pts[-1] else pts


def reduzir(origem: Path, destino: Path, lado: int = 1600) -> None:
    Image.MAX_IMAGE_PIXELS = None
    img = Image.open(origem).convert("RGB")
    img.thumbnail((lado, lado))
    destino.parent.mkdir(parents=True, exist_ok=True)
    img.save(destino, quality=82)


def faltantes(curadoria: dict, raiz: Path) -> list[str]:
    return [f["arquivo"] for m in curadoria.values() for f in m["fotos"] if not (raiz / f["arquivo"]).exists()]


def main() -> int:
    sys.stdout.reconfigure(encoding="utf-8")
    cfg = MarcosCfg.carregar()
    raiz = terreno_dir()
    curadoria = json.loads(CURADORIA.read_text(encoding="utf-8"))
    if falta := faltantes(curadoria, raiz):
        raise SystemExit("fotos da curadoria que não existem:\n" + "\n".join(falta))
    georref = json.loads((RAIZ / "dados" / "georref.json").read_text(encoding="utf-8"))
    proj = Projecao(georref["origem"]["E"], georref["origem"]["N"])
    encaixe = json.loads(MARCOS_JSON.read_text(encoding="utf-8"))
    with zipfile.ZipFile(raiz / cfg.kmz_estande) as z:
        kml = z.read(next(n for n in z.namelist() if n.endswith(".kml"))).decode("utf-8")
        planta = z.read(PLANTA_KMZ)
    estande = Polygon([proj.xy(lon, lat) for lon, lat in estande_do_kmz(kml)])
    shutil.rmtree(FOTOS, ignore_errors=True)
    saida = []
    for id_, m in curadoria.items():
        if m["objeto"]:
            e = encaixe["marcos"][m["objeto"]]
            c = Polygon(e["pegada"]).centroid
            x, y, z = c.x, c.y, e["z"]
        else:
            c = estande.centroid
            x, y, z = c.x, c.y, encaixe["cota"]
        lon, lat = proj.lonlat(x, y)
        galeria = []
        for k, f in enumerate(m["fotos"], 1):
            nome = f"{id_}/{k:02d}.jpg"
            reduzir(raiz / f["arquivo"], FOTOS / nome)
            galeria.append({"foto": f"marcos/{nome}", "legenda": f["legenda"]})
        item = {"id": id_, "nome": m["nome"], "x": round(x, 2), "y": round(y, 2), "z": round(z, 2),
                "objeto": m["objeto"], "galeria": galeria, "planta": None,
                "lat": round(lat, 7), "lon": round(lon, 7)}
        if id_ == "estande":
            (FOTOS / "estande").mkdir(parents=True, exist_ok=True)
            (FOTOS / "estande" / "planta.jpg").write_bytes(planta)
            reduzir(FOTOS / "estande" / "planta.jpg", FOTOS / "estande" / "planta.jpg")
            item["planta"] = "marcos/estande/planta.jpg"
            item["pegada"] = [[round(px, 2), round(py, 2)] for px, py in estande.exterior.coords[:-1]]
        saida.append(item)
    DESTINO.write_text(json.dumps({"versao": 1, "marcos": saida}, ensure_ascii=False, indent=1), encoding="utf-8")
    mb = sum(p.stat().st_size for p in FOTOS.rglob("*.jpg")) / 1e6
    print(f"{len(saida)} marcos · {sum(len(m['galeria']) for m in saida)} fotos · {mb:.1f} MB em dados/marcos/")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
